/* ---------- Cart actions ---------- */
function addToCart(id){
  const existing = state.cart.find(c=>c.id===id);
  if(existing){ existing.qty++; } else { state.cart.push({id, qty:1}); }
  saveCart();
  showToast(t('toast_added_cart'));
  render();
}
function removeFromCart(id){
  state.cart = state.cart.filter(c=>c.id!==id);
  saveCart();
  showToast(t('toast_removed_cart'));
  render();
}
function changeQty(id, delta){
  const item = state.cart.find(c=>c.id===id);
  if(!item) return;
  item.qty += delta;
  if(item.qty<=0){ removeFromCart(id); return; }
  saveCart();
  render();
}
function cartCount(){ return state.cart.reduce((s,c)=>s+c.qty,0); }

  const h = hashStr(fileNameSeed + Date.now());
  if(tool==='disease'){
    const d = DISEASES[h % DISEASES.length];

    return { kind:'disease', name:d, confidence, severity:d.sev, rec:DISEASE_REC[d.badge] };
  }
  if(tool==='ripeness'){
    const r = RIPENESS_STAGES[h % RIPENESE_LEN()];
    const confidence = 74 + (h % 24);
    const sugar = (8 + (h % 10)) + 'â€“' + (10 + (h % 10)) + ' Â°Brix';
    return { kind:'ripeness', stage:r, confidence, sugar };
  }
  if(tool==='quality'){
    const g = GRADES[h % GRADES.length];
    const confidence = 80 + (h % 18);
    const uniformity = 70 + (h % 28);
    const surface = 68 + (h % 30);

  }

function filteredProduce(){
  const f = state.marketFilters;
  const lang = state.lang;
  let list = PRODUCE.filter(p=>{
    if(f.category!=='all' && p.category!==f.category) return false;
    if(f.location!=='all' && p.location!==f.location) return false;
    if(f.grade!=='all' && p.grade!==f.grade) return false;
    if(f.q){
      const q = f.q.toLowerCase();
      const hay = (p.name.en+' '+p.name.hi+' '+p.farmer+' '+p.location).toLowerCase();
      if(!hay.includes(q)) return false;
    }
    return true;
  });
  if(f.sort==='price_low') list = list.slice().sort((a,b)=>a.price-b.price);
  else if(f.sort==='price_high') list = list.slice().sort((a,b)=>b.price-a.price);
  else if(f.sort==='grade') list = list.slice().sort((a,b)=>a.grade.localeCompare(b.grade));
  else list = list.slice().sort((a,b)=>a.days-b.days);
  return list;
}

function fileToDataUrl(file){
  return new Promise((resolve,reject)=>{
    const r = new FileReader();
    r.onload = ()=>resolve(r.result);
    r.onerror = reject;
    r.readAsDataURL(file);
  });
}

async function processFile(file, tool){
  if(!file) return;
  if(!file.type.match(/image\/(png|jpeg)/)){ showToast('Please upload a JPG or PNG image.'); return; }
  if(file.size > 8*1024*1024){ showToast('Image is larger than 8 MB.'); return; }
  let dataUrl;
  try{ dataUrl = await fileToDataUrl(file); }catch(e){ return; }

  state.aiLoading[tool] = true;
  state.aiCurrent[tool] = { id:'x', thumb:dataUrl, result:null };
  render();

  setTimeout(()=>{
    const result = runAnalysis(tool, file.name+file.size);
    const id = 'a'+Date.now();
    const entry = { id, thumb:dataUrl, result };
    state.aiHistory[tool] = [entry, ...state.aiHistory[tool]].slice(0,6);
    state.aiCurrent[tool] = entry;
    state.aiLoading[tool] = false;
    render();
  }, 1100 + Math.random()*700);
}
function handleFileInput(e, tool){ processFile(e.target.files[0], tool); }
function handleDrop(e, tool){
  e.preventDefault();
  e.currentTarget.classList.remove('drag');
  const file = e.dataTransfer.files[0];
  processFile(file, tool);
}
function selectHistory(tool, id){
  const entry = state.aiHistory[tool].find(h=>h.id===id);
  if(entry){ state.aiCurrent[tool] = entry; render(); }
}
